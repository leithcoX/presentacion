\begin{frame}

	Se pide
	\begin{itemize}
		\item Para cada tramo de la trayectoria, realizar el DCL para el cilindro, plantear las ecuaciones dinámicas de movimiento, indicando claramente el sistema de referencia y el sistema de coordenadas elegidos.
		\item Calcular la aceleración angular y la del centro de masa del cilindro durante la subida en el tramo $\overline{AB}$ y la aceleración angular en el tramo liso.

	\end{itemize}

	Analicemos qué fuerzas realizan trabajo
	\begin{columns}

		\end{figure}
		\column{0.5\textwidth}
		% $$ \vec{F} = F\,(\cos(\alpha)\,\uveci + \sin(\alpha)\,\uvecj )\ \ \text{ y } \ \ m\,\vec{g} =  $$
		$$ m\,\vec{g} = m\,g\,(\sin(\alpha) \uveci + \cos(\alpha) \uvecj) $$
		$$ \vec{F} = F\,(\cos(\alpha)\,\uveci + \sin(\alpha)\,\uvecj )$$
		$$ \vec{N} = N\, \uvecj \ \ \ \text{ y } \ \ \ \vec{\fr}_e = -\fre\, \uveci$$
		\centering

		Y además
		$$ \mathrm{d}\vec{x} = \mathrm{d}x\,\uveci \implies \uveci \int_{x_0}^{x_f} \mathrm{d}x= \Delta x \, \uveci $$
	\end{columns}

	$$ W_{\vec{N}} = N\,\uvecj \cdot \Delta x\, \uveci  = 0$$
	$$ W_{\vec{\fre}_e} = -\fre\,\uveci \cdot \Delta x\, \uveci  = 0$$
	$$ W_{\vec{F}} =F\,(\cos(\alpha)\,\uveci + \sin(\alpha)\,\uvecj ) \cdot \Delta x \, \uveci $$
	$$ W_{m\vec{g}} =m\,g\,(\sin(\alpha)\,\uveci + \cos(\alpha)\,\uvecj ) \cdot \Delta x \, \uveci $$
\end{frame}

\begin{frame}
	\frametitle{Conservación de la Energía}
	Planteamos los teoremas de conservación de la energía
	$$ \Delta EC= EC_f - EC_o = WF_{NC} + WF_C $$
	$$ \Delta EP = - WF_C $$
	$$ \Delta EM = \Delta EC + \Delta EP = WF_{NC} $$

	Como el cuerpo parte del reposo
	$$ \Delta EC = EC_f $$

	Y la única fuerza no conservativa que realiza trabajo es $\vec{F}$
	$$ \Delta EM = EC_f + \Delta EP  = W_{\vec{F}}$$
\end{frame}

\begin{frame}
	\frametitle{Conservación de la Energía}
	\framesubtitle{Velocidad angular del CR}

	Determinamos la velocidad angular $\omega$ en el instante en el que deja de haber rozamiento.
	$$  \vec{v}_{\tcm} = \vec{v}_{\text{\tiny CIR}} + \vec{v}_{\tcm / \text{\tiny CIR}} = \vec{\omega} \times \vec{r}_{\tcm / \text{\tiny CIR}}$$
	$$\left| \begin{matrix}
			\uveci & \uvecj & \hat{k} \\
			0      & 0      & \omega  \\
			0      & -R     & 0
		\end{matrix}\right| $$
	$$ \vec{v}_\tcm = \omega\,R\, \uveci \implies \omega = \frac{v_\tcm}{R} $$

	Y como se demostró en~\eqref{eq:torque_conservation}, $\gamma = 0 \implies \omega = \text{cte.}$ durante todo el tramo $\overline{BC}$.


\end{frame}

\begin{frame}
	\frametitle{Conservación de la Energía}
	Tomando las definiciones de $EC$ y $EP$
	% $$ EC_f = \frac{1}{2}\,m\,v_{\tcm_f}^2 + \frac{1}{2}\, I_\tcm \,\omega^2 \ \ \text{y} \ \ \Delta EP = m\,g\,h_{\tcm_C} - m\,g\,h_{\tcm_A}  $$
	$$ EC_f = \frac{1}{2}\,m\,v_{\tcm_f}^2 + \frac{1}{2}\, I_\tcm \,\omega^2 $$
	$$\Delta EP = m\,g\,h_{\tcm_C} - m\,g\,h_{\tcm_A}  $$


	Y sabiendo que

	$$ \sin(\alpha) = \frac{h_{\tcm_C} - h_{\tcm_A}}{\Delta x} \implies \Delta x = \frac{h_{\tcm_C} - h_{\tcm_A}}{\sin(\alpha)}$$

	\vspace{1em}
	Desarrollamos la ecuación
	% $$ \frac{1}{2}\,m\,v_{\tcm_f}^2 + \frac{1}{4}\,m\,R^2\,\left( \frac{v_{\tcm_f}}{R}\right)^2  + m\,g\,\left(h_{\tcm_C} - h_{\tcm_A} \right) = \Delta x \, F \cos(\alpha)$$
	$$ \frac{1}{2}\,m\,v_{\tcm_f}^2 + \frac{1}{4}\,m\,R^2\,\left( \frac{v_{\tcm_f}}{R}\right)^2  + m\,g\,\left(h_{\tcm_C} - h_{\tcm_A} \right) = \frac{h_{\tcm_C} - h_{\tcm_A}}{\tan(\alpha)}F $$

	$$ \frac{3}{4}\,m\,v_{\tcm_f}^2 = (F\, \cot(\alpha) - m\,g)\,\left(h_{\tcm_C} - h_{\tcm_A} \right) $$

\end{frame}

\begin{frame}
	\frametitle{Conservación de la Energía}
	Teniendo en cuenta los datos del ejercicio
	$$ \begin{array}{c c c}
			F = \qty{20}{\newton} & m = \qty{2}{\kilo\gram} & g = \qty[per-mode=fraction]{10}{\meter\per\second\squared} \\ \alpha = \qty{30}{\degree} & h_{\tcm_A} = \qty{1}{\meter} & h_{\tcm_C} = \qty{4}{\meter}
		\end{array} $$

	$$ v_{\tcm} = \sqrt{\frac{F\, \cot(\alpha) - m\,g}{3\,m}\,4\,\left(h_{\tcm_C} - h_{\tcm_A} \right)} $$
	$$ v_{\tcm} = \sqrt{\frac{\qty{20}{\newton}\cdot \cot(\qty{30}{\degree}) - \qty{2}{\kilo\gram}\cdot\qty[per-mode=fraction]{10}{\meter\per\second\squared}}{3\cdot\qty{2}{\kilo\gram}}\cdot4\cdot\left(\qty{4}{\meter} - \qty{1}{\meter}\right)}$$

	$$ v_{\tcm_C} \approx \qty[per-mode = fraction]{5.41}{\meter\per\second} $$
	$$ \vec{v}_{\tcm_C} \approx \qty[per-mode = fraction]{5.41}{\meter\per\second} \, \uveci $$

\end{frame}
